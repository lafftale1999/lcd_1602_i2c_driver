37static uint8_t write_nibble(i2c_master_dev_handle_t handle, uint8_t nibble, bool rs) {
38 uint8_t data = (nibble & 0xF0);
39
40 if(rs) data |= LCD_1602_RS;
41
42 data |= LCD_1602_BACKLIGHT;
43
44 uint8_t sequence[2];
45 sequence[0] = data | LCD_1602_ENABLE;
46 sequence[1] = data & ~LCD_1602_ENABLE;
47
48 /* i2c_master_transmit() expects the timeout in milliseconds, not ticks */
49 esp_err_t ret = i2c_master_transmit(handle, sequence, sizeof(sequence), I2C_MASTER_TIMEOUT_MS);
50
51 return (ret == ESP_OK) ? 0 : 1;
52}

63static uint8_t write_byte(i2c_master_dev_handle_t handle, uint8_t byte, bool rs) {
64 uint8_t err = 0;
65
66 uint8_t high = byte & 0xF0;
67 uint8_t low = (byte << 4) & 0xF0;
68
69 err |= write_nibble(handle, high, rs);
70 err |= write_nibble(handle, low, rs);
71 lcd_delay_ms(1);
72
73 return err;

127static LCD_WRITE_STATUS write_row(i2c_master_dev_handle_t handle, uint8_t row, const char *str) {
128 if(str == NULL) str = "";
129
130 uint8_t err = lcd_goto(handle, 0, row);
131
132 uint8_t col = 0;
133 while(col < LCD_1602_SCREEN_CHAR_WIDTH && *str != '\0' && *str != '\n') {
134 err |= lcd_1602_send_char(handle, *str);
135 str++;
136 col++;
137 }
138
139 while(col < LCD_1602_SCREEN_CHAR_WIDTH) {
140 err |= lcd_1602_send_char(handle, ' ');
141 col++;
142 }
143
144 if(err) return LCD_WRITE_INTERRUPTED;
145
146 /* Anything left except a single trailing newline did not fit */
147 if(*str == '\n') str++;
148 return (*str == '\0') ? LCD_WRITE_FINISHED : LCD_TOO_LONG_STRING;
149}

159LCD_WRITE_STATUS lcd_1602_send_string(i2c_master_dev_handle_t handle, const char *str) {
160 if(str == NULL) return LCD_WRITE_FINISHED;
161
162 uint8_t err = lcd_1602_clear_screen(handle);
163
164 uint8_t char_len = 0;
165 uint8_t row = 0;
166
167 while(*str != '\0') {
168 if(*str == '\n' || char_len >= LCD_1602_SCREEN_CHAR_WIDTH) {
169 /* A newline that ends the string needs no new row */
170 if(*str == '\n' && *(str + 1) == '\0') {
171 break;
172 }
173
174 if(row + 1 < LCD_1602_MAX_ROWS) {
175 row++;
176 err |= lcd_goto(handle, 0, row);
177 char_len = 0;
178
179 if(*str == '\n') {
180 str++;
181 continue;
182 }
183 }
184 else return LCD_WRITE_INTERRUPTED;
185 }
186
187 err |= lcd_1602_send_char(handle, *str);
188 char_len++;
189 str++;
190 }
191
192 return err ? LCD_WRITE_INTERRUPTED : LCD_WRITE_FINISHED;
193}

195uint8_t lcd_1602_init(i2c_master_dev_handle_t handle) {
196/*
197This function sets up the standard mode of the LCD and follows
198a specific start up sequence described by the manufacturer.
199*/
200 uint8_t err = 0;
201
202 // Manufacturer specific order
203 lcd_delay_ms(50); // Wait >40 ms after power on
204 err |= write_nibble(handle, (0x03 << 4), false);
205 lcd_delay_ms(5); // Wait >4.1 ms
206 err |= write_nibble(handle, (0x03 << 4), false);
207 lcd_delay_ms(1); // Wait >100 us
208 err |= write_nibble(handle, (0x03 << 4), false);
209 lcd_delay_ms(1);
210 err |= write_nibble(handle, (0x02 << 4), false);
211 lcd_delay_ms(1);
212 // End of manufacturer specific order
213
214 // 5x10 font is only supported in single-row mode, so 2 rows must use 5x8
215 err |= send_command(handle, LCD_1602_FUNCTION_SET(LCD_1602_DATA_LEN_4_BIT, LCD_1602_2_ROWS, LCD_1602_FONT_5X7));
216 err |= send_command(handle, LCD_1602_CONFIG_DISPLAY_SWITCH(LCD_1602_DISPLAY_ON, LCD_1602_CURSOR_OFF, LCD_1602_N_BLINK_DISPLAY));
217 err |= lcd_1602_clear_screen(handle);
218 err |= send_command(handle, LCD_1602_CONFIG_INPUT_SET(LCD_1602_INCREMENT_MODE, LCD_1602_CURSOR_N_MOVE));
219
220 return err ? 1 : 0;
221}
